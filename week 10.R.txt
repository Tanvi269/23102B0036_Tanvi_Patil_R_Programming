# ---------------------------------------------------------------------
# 1. String Substitution and Replacement (sub and gsub)
# ---------------------------------------------------------------------

# Replace the first instance of a substring
y_sub <- "Number of participants: 25"
sub("25", "30", y_sub)

# Replace substrings (sub replaces first instance, gsub replaces all)
y_rep <- "Mr. Singh is the smart one. Mr. Singh is funny, too."
sub("Mr. Singh", "Professor Jha", y_rep)   # First occurrence only
gsub("Mr. Singh", "Professor Jha", y_rep)  # All occurrences


# ---------------------------------------------------------------------
# 2. String Searching (grep and grepl)
# ---------------------------------------------------------------------

str_vec <- c("R Course", "exercises", "include examples of R language")

# Return matching values or indices
grep("ex", str_vec, value = TRUE)
grep("ex", str_vec, value = FALSE)

# Case-sensitive and case-insensitive matching
str_case <- c("R Course", "exercises", "include examples of r language", "in R software.")
grep("R", str_case, ignore.case = TRUE, value = TRUE)
grep("R", str_case, ignore.case = FALSE, value = FALSE)

# Searching across combined vector elements
x_date <- "R course 24.07.2021"
y_part <- "Number of participants: 25"
grep("our", c(x_date, y_part))

# Returning logical vectors using grepl
grepl("R", str_vec)
grepl("ex", str_vec)

# ---------------------------------------------------------------------
# 5. Combining and Merging Data Frames
# ---------------------------------------------------------------------

# Sample data frames for joining
df1 <- data.frame(state = c("UP", "MP", "AP", "JK"), popnsize = c(1000, 2000, 3000, 4000))
df2 <- data.frame(state = c("UP", "MP", "AP", "JK"), samplesize = c(100, 200, 300, 400), surveycompleted = c("Yes", "No", "Yes", "No"))

# Horizontal combinations
cbind(df1, df2)
merge(df1, df2, by = "state")

# Vertical stacking
df11 <- data.frame(state = c("UP", "MP", "AP", "JK"), popnsize = c(1000, 2000, 3000, 4000))
df22 <- data.frame(state = c("Bihar", "Delhi", "Punjab"), popnsize = c(100, 200, 300))
rbind(df11, df22)
