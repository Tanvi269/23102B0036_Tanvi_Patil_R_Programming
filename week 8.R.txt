# ------------------------------------------------------------------------------
# Lecture 31: Operations on Lists
# ------------------------------------------------------------------------------

# Merging Lists
list1 = list(1, 2, 3)
list2 = list("water", "juice", "lemonade")
list12 = c(list1, list2)
list12

# Converting Lists to Vectors
unlist(list1)
unlist(list2)
mode(list1)
mode(unlist(list1))

# Appending Lists
append(list1, 100)
append(list2, "coffee")
append(list1, 100, after = 2)
append(list2, "coffee", after = 2)

# Removing from Lists
list1[-2]
list2[-1]

# Extracting from Lists
list1_ext = list(1, 2, 3, 4, 5, 6)
list2_ext = list("water", "juice", "lemonade", "tea", "coffee", "milk")
list1_ext[2:4]
list1_ext[c(1, 3, 5)]
list2_ext[2:4]
list2_ext[c(1, 3, 5)]


# ------------------------------------------------------------------------------
# Lecture 32: Vector Indexing
# ------------------------------------------------------------------------------

# Vector and Conditional Indexing
x = 1:10
x[(x > 5)]
x[(x %% 2 == 0)] # x mod 2 is 0 (even numbers)
x[(x %% 2 == 1)] # x mod 2 is 1 (odd numbers)

# Missing Observations
x[5] = NA
x
y = x[!is.na(x)] # ! means negation
y
mean(x) # Returns NA
mean(y)

# Negative Integer Indexing
x = 1:10
x[-(1:5)]
x[(6:10)]

# String Vectors and Named Elements
z = list(a1 = 1, a2 = "c", a3 = 1:3)
names(z)
names(z)[3] = "c2"
z

x_named = c(water = 1, juice = 2, lemonade = 3)
names(x_named)
x_named["juice"]

# Empty Index & Mixed Mode Matrices
x = 1:10
x[]

ab = list(1, 2, 3, "X", "Y", "Z")
dim(ab) = c(2, 3)
print(ab)

# ------------------------------------------------------------------------------
# Lecture 33: Factors - Creation
# ------------------------------------------------------------------------------

y = c(1, 4, 3, 5, 4, 2, 4)
possible.dieface = c(1, 2, 3, 4, 5, 6)
labels.dieface = c("one", "two", "three", "four", "five", "six")
facy = factor(y, levels = possible.dieface, labels = labels.dieface)
facy


# ------------------------------------------------------------------------------
# Lecture 34: Factors - Class and Unclass
# ------------------------------------------------------------------------------

# Coercion to Factor
x_vec = c(3, 4, 5, 6, 1, 2, 3, 3, 4, 4, 5, 6)
y_fac = as.factor(x_vec)
y_fac

x_str = factor(c("lemonade", "lemonade", "juice", "lemonade", "water"))
x_str

# Class Function
class(9)
class("9")
class(print)
x_mat = matrix(nrow = 2, ncol = 2, data = 1:4)
class(x_mat)

# Unclass Function & Levels
brands = c("A", "A", "B", "B", "B", "B", "C")
brands_fac = factor(brands)
brands_fac
unclass(brands_fac)

colours = c("blue", "green", "red")
colours[unclass(brands_fac)]

x_drink = factor(c("lemonade", "lemonade", "juice", "lemonade", "water"))
unclass(x_drink)

# Custom Level Assignment
x_custom = factor(c("lemonade", "lemonade", "juice", "lemonade", "water"), 
                  levels = c("water", "juice", "lemonade"))
x_custom
unclass(x_custom)
levels(x_custom)

# Ordered Factors
income = ordered(c("high", "high", "low", "medium", "medium"), 
                 levels = c("low", "medium", "high"))
income
unclass(income)

# Print & Precision Formatting

print("apple")

# Format Width & Justification

format(c("A", "BB", "CCC", "DDDD"), width = 7, justify = "left")
format(c("A", "BB", "CCC", "DDDD"), width = 7, justify = "right")
format(c("A", "BB", "CCC", "DDDD"), width = 7, justify = "none")
