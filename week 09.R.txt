# ==========================================
# 1. PRINT AND CONCATENATE (print vs cat)
# ==========================================

# Limitation of print() with multiple arguments (returns an error)[cite: 3]
# print("The zero occurs at", 2 * pi, "radians.")

# Printing multiple items one at a time[cite: 4, 8]
print("The zero occurs at")
print(2 * pi)
print("radians")

# Using the cat() function for continuous output[cite: 8]
cat("The zero occurs at", 2 * pi, "radians.", "\n")

# Working with dates[cite: 10]
d <- date()
cat("Today's date is:", d, "\n")

# Using the sep parameter in cat()[cite: 11, 12]
x <- 1:10
cat(x, sep = "++")
cat("\n") # Newline character
cat(x, sep = " / ")
cat("\n")

# Formatting numbers and strings inside cat()[cite: 13, 14]
x_val <- 7
cat("The square of", x_val, "is", x_val^2, "!\n")
cat("The square root of", x_val, "is approximately", format(sqrt(x_val), digits = 3), "\n")

# Printing simple vectors with cat()[cite: 15]
evenno <- c(2, 4, 6, 8, 10)
cat("The first few even numbers are:", evenno, "...\n")

# Using fill and labels parameters in cat()[cite: 16]
cat(1:10, fill = 2, labels = paste("(", letters[1:10], "):"))


# ==========================================
# 2. PASTE FUNCTION (paste and paste0)
# ==========================================

# Converting vectors to character strings[cite: 22]
paste(1:12)
as.character(1:12)

# Using custom separators[cite: 23, 24]
paste("Everybody", "loves", "R Programming.")
paste("Everybody", "loves", "R Programming.", sep = "*")
paste("Everybody", "loves", "R Programming.", sep = "===")

# Generating combinations from string vectors[cite: 25]
names <- c("Prof. Singh", "Mr. Venkat", "Dr. Jha")
paste(names, "is", "a good", "person.")

# Using the collapse parameter[cite: 26]
paste(names, "is", "a good", "person.", collapse = ", and ")

# Handling spaces and custom separators[cite: 28]
paste(1, " is first", 2, " is second", 3, " is third", sep = "#")

# Creating indexed strings[cite: 29]
x_ex <- paste("Ex", 1:5, sep = "_")
x_ex[1]
x_ex[2]
x_ex[3]
x_ex[4]
x_ex[5]

# Combining sep and collapse[cite: 30, 31]
x_collapsed <- paste("Ex", 1:5, sep = "_", collapse = "")

# Comparing paste0 and paste[cite: 32]
paste0(1:10)
paste(1:10)

# Vectorized concatenation with suffixes[cite: 33]
paste0(1:10, c("st", "nd", "rd", rep("th", 7)))
paste(1:10, c("st", "nd", "rd", rep("th", 7)))

# 3. STRING SPLITTING (strsplit)
# ==========================================

x_str <- "The&! syntax&!of&!paste&!is! &available! &inthe online-help"

# Splitting using a single character[cite: 36, 37]
strsplit(x_str, split = "!")

# Splitting using a substring pattern[cite: 38, 39, 40]
strsplit(x_str, split = "&!")
y_split <- strsplit(x_str, split = "!&")

# Accessing split components[cite: 41]
y_split[[1]][1]
y_split[[1]][2]
y_split[[1]][3]

# Converting split strings into matrices[cite: 43, 44, 45, 46]
dates <- c("2020-07-24", "2021-08-25", "2022-09-26", "2023-10-27")
datesplt <- strsplit(dates, "-")

# Character matrix
datemat <- matrix(unlist(datesplt), nrow = 4, ncol = 3, byrow = TRUE)
datemat

# Numeric matrix
datematrix <- matrix(as.numeric(unlist(datesplt)), nrow = 4, ncol = 3, byrow = TRUE)
datematrix

# Splitting every character of a string[cite: 47]
strsplit("Shalabh", split = "")
